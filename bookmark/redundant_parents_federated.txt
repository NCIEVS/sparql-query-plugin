# finds & reports redundant parents in multi-parent concepts
# set the timeout to max 100+

PREFIX ncit: <http://ncicb.nci.nih.gov/xml/owl/EVS/Thesaurus.owl#>

SELECT distinct
  (STR(?lcode) AS ?code) 
  (STR(?llabel) AS ?label) 
  (STR(?lsem_type) AS ?sem_type) 
  (STR(?lred_code) AS ?red_code) 
  (STR(?lredundant) AS ?redundant) 
  (STR(?lnonred_code) AS ?nonred_code) 
  (STR(?lnonredundant) AS ?nonredundant)
WHERE {
  ?s ncit:NHC0 ?lcode ;
     rdfs:label ?llabel .
  OPTIONAL { ?s ncit:P106 ?lsem_type }

  OPTIONAL { 
    ?o2 ncit:NHC0 ?lred_code ;
        rdfs:label ?lredundant 
  }
  OPTIONAL { 
    ?o3 ncit:NHC0 ?lnonred_code ;
        rdfs:label ?lnonredundant 
  }

   {
      ?s rdfs:subClassOf ?o2 .
      OPTIONAL { ?s rdfs:subClassOf ?o3 }
      OPTIONAL { ?o3 rdfs:subClassOf+ ?o2 }
    }
  }

# ORDER BY ?label
limit 1000
